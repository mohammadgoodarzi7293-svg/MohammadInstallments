/* ================================ اصلاح نهایی مانده وام + سطل زباله ================================ */ function getRealMonths(item) { let months = Number(item.months || 0); if (months >= 1) return months; const start = String(item.startDate || "") .split("/") .map(Number); const end = String(item.endDate || "") .split("/") .map(Number); if ( start.length === 3 && end.length === 3 && start.every(Boolean) && end.every(Boolean) ) { const result = (end[0] - start[0]) * 12 + (end[1] - start[1]) + 1; if (result > 0) return result; } return 0; } /* تعداد کارمزدهای سالانه مثال: 12 قسط → 0 کارمزد 13 قسط → 1 کارمزد 24 قسط → 1 کارمزد 25 قسط → 2 کارمزد */ function getAnnualFeeCount(item) { const months = getRealMonths(item);
